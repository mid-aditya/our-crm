[DATABASE SCHEMA - SUPABASE]

-- 1. PROFILES (extend auth.users)
create table profiles (
  id uuid references auth.users primary key,
  full_name text not null,
  phone text unique,
  role text check (role in ('owner', 'admin', 'sales', 'support')),
  team_id uuid references teams(id),
  avatar_url text,
  created_at timestamptz default now()
);

-- 2. TEAMS (multi-tenant support)
create table teams (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  whatsapp_business_id text,
  settings jsonb default '{}',
  created_at timestamptz default now()
);

-- 3. CONTACTS
create table contacts (
  id uuid primary key default gen_random_uuid(),
  team_id uuid references teams(id) not null,
  name text not null,
  whatsapp_number text, -- format: +628xxx
  email text,
  label text[], -- ['vip', 'prospect', 'customer']
  source text, -- 'broadcast', 'inbox', 'import'
  notes text,
  last_activity_at timestamptz,
  created_at timestamptz default now(),
  unique(team_id, whatsapp_number)
);

-- 4. DEALS (Sales Pipeline)
create table deals (
  id uuid primary key default gen_random_uuid(),
  team_id uuid references teams(id) not null,
  contact_id uuid references contacts(id),
  title text not null,
  value numeric,
  currency text default 'IDR',
  stage text check (stage in ('chat_masuk', 'tertarik', 'ditawar', 'deal', 'batal')),
  assigned_to uuid references profiles(id),
  reminder_at timestamptz,
  closed_at timestamptz,
  created_at timestamptz default now()
);

-- 5. TASKS (Ticketing / To-Do)
create table tasks (
  id uuid primary key default gen_random_uuid(),
  team_id uuid references teams(id) not null,
  contact_id uuid references contacts(id),
  deal_id uuid references deals(id),
  title text not null,
  description text,
  priority text check (priority in ('low', 'medium', 'urgent')),
  status text check (status in ('todo', 'in_progress', 'done', 'cancelled')),
  due_date timestamptz,
  assigned_to uuid references profiles(id),
  completed_at timestamptz,
  created_at timestamptz default now()
);

-- 6. CAMPAIGNS (Broadcast Management)
create table campaigns (
  id uuid primary key default gen_random_uuid(),
  team_id uuid references teams(id) not null,
  name text not null,
  channel text check (channel in ('whatsapp', 'email', 'sms', 'rcs')),
  message_template text,
  audience_filter jsonb, -- {labels: ['vip'], min_value: 100000}
  scheduled_at timestamptz,
  status text check (status in ('draft', 'scheduled', 'sending', 'completed', 'failed')),
  stats jsonb default '{"sent": 0, "delivered": 0, "failed": 0, "replied": 0}',
  created_by uuid references profiles(id),
  created_at timestamptz default now()
);

-- 7. MESSAGES (Unified Inbox)
create table messages (
  id uuid primary key default gen_random_uuid(),
  team_id uuid references teams(id) not null,
  contact_id uuid references contacts(id),
  direction text check (direction in ('inbound', 'outbound')),
  channel text check (channel in ('whatsapp', 'email', 'sms')),
  content text not null,
  media_url text[],
  status text check (status in ('sent', 'delivered', 'read', 'failed')),
  external_message_id text, -- ID dari WhatsApp API
  created_at timestamptz default now()
);

-- 8. ACTIVITY_LOGS (Audit Trail)
create table activity_logs (
  id uuid primary key default gen_random_uuid(),
  team_id uuid references teams(id) not null,
  actor_id uuid references profiles(id),
  action text not null, -- 'contact_created', 'deal_updated', etc
  entity_type text,
  entity_id uuid,
  details jsonb,
  created_at timestamptz default now()
);

-- INDEXES untuk performa
create index idx_contacts_team on contacts(team_id);
create index idx_contacts_wa on contacts(whatsapp_number) where whatsapp_number is not null;
create index idx_deals_pipeline on deals(team_id, stage);
create index idx_tasks_due on tasks(team_id, due_date) where status != 'done';
create index idx_messages_contact on messages(contact_id, created_at desc);


[ROW LEVEL SECURITY (RLS) - WAJIB!]

-- Enable RLS di semua tabel
alter table contacts enable row level security;
alter table deals enable row level security;
-- ... ulangi untuk semua tabel

-- Policy: User hanya akses data tim-nya sendiri
create policy "Team isolation on contacts"
  on contacts for all
  using (team_id in (
    select team_id from profiles where id = auth.uid()
  ));

-- Policy: Sales hanya lihat contacts yang di-assign (opsional, lebih ketat)
create policy "Sales see assigned contacts"
  on contacts for select
  using (
    auth.uid() in (
      select id from profiles 
      where team_id = contacts.team_id 
      and role in ('owner', 'admin', 'sales')
    )
  );

-- Policy: Owner/admin bisa full access dalam tim-nya
create policy "Admin full access"
  on contacts for all
  using (
    exists (
      select 1 from profiles 
      where id = auth.uid() 
      and team_id = contacts.team_id 
      and role in ('owner', 'admin')
    )
  );

// src/lib/supabase/queries.ts
import { createClient } from './client'

export const contacts = {
  getAll: async (teamId: string, filters?: { label?: string; search?: string }) => {
    const supabase = createClient()
    let query = supabase.from('contacts').select('*').eq('team_id', teamId)
    
    if (filters?.label) query = query.contains('label', [filters.label])
    if (filters?.search) query = query.ilike('name', `%${filters.search}%`)
    
    return query.order('created_at', { ascending: false })
  },
  
  create: async (data: { team_id: string; name: string; whatsapp_number: string }) => {
    const supabase = createClient()
    // Format phone number ke +62
    const wa = data.whatsapp_number.startsWith('08') 
      ? '+62' + data.whatsapp_number.slice(1) 
      : data.whatsapp_number
      
    return supabase.from('contacts').insert({ ...data, whatsapp_number: wa }).select().single()
  },
  
  // ... update, delete, import
}

// src/app/(dashboard)/contacts/page.tsx (simplified)
'use client'
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
import { contacts } from '@/lib/supabase/queries'

export default function ContactsPage({ teamId }: { teamId: string }) {
  const queryClient = useQueryClient()
  
  const { data: contactsList, isLoading } = useQuery({
    queryKey: ['contacts', teamId],
    queryFn: () => contacts.getAll(teamId)
  })
  
  const createMutation = useMutation({
    mutationFn: (data: any) => contacts.create(data),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['contacts', teamId] })
      // show toast success
    }
  })
  
  if (isLoading) return <ContactsSkeleton />
  
  return (
    <div>
      <ContactForm onSubmit={(data) => createMutation.mutate({ ...data, team_id: teamId })} />
      <ContactList contacts={contactsList?.data || []} />
    </div>
  )
}
